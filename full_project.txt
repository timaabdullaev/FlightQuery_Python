# Flight Schedule Parser and Query Tool: starter files

This folder is the starting point for the graded lab. The full rules are in
the assignment description on Moodle; this file only explains what is here.

## What is in this folder

| Path | What it is |
| --- | --- |
| `flight_parser.py` | Entry point with `main()` and your student constants. **Edit.** |
| `validator.py` | `load_airports`, `validate_row`, `to_record`. **Edit.** |
| `csv_loader.py` | `parse_files`. **Edit.** |
| `query.py` | `run_query`, `run_queries`. **Edit.** |
| `AI_USAGE.md` | Your AI use disclosure. **Fill in before submitting.** |
| `data/db.csv` | The sample file from the worked example (7 valid rows, 12 invalid). |
| `data/airports.txt` | The airport list to pass with `-a`. |
| `data/query.json` | The sample queries from the worked example. |
| `data/flights/` | A folder for testing `-d`. One of its files uses Windows line endings, one has only a header, and the `.txt` file and the `archive/` sub-folder must be ignored. |
| `tests/run_public_tests.py` | The public test script. Do not edit. |
| `tests/expected/` | Expected outputs used by the tests. Do not edit. |
| `.devcontainer/` | Codespaces setup (Python 3.13). Do not edit. |

The function names and signatures in the four `.py` files are fixed, because
the autograder calls them directly. Replace the `raise NotImplementedError`
lines with your code. You may add more functions, classes and modules.

## Getting started

1. Create your own **private** repository from the starter template and
   open it in GitHub Codespaces. The `.devcontainer` folder sets the
   codespace up with Python 3.13; check with `python --version`.
2. Add your instructor as a collaborator (Settings > Collaborators) so the
   grader can read your repository.
3. Set `STUDENT_ID`, `FIRST_NAME` and `LAST_NAME` in `flight_parser.py`.
4. Start with `validator.py`; the tests in group B check it on its own.
5. Run your program from the project folder, for example:

   ```
   python flight_parser.py -i data/db.csv -a data/airports.txt
   python flight_parser.py -d data/flights -a data/airports.txt -o out/db.json
   python flight_parser.py -j db.json -q data/query.json
   ```

   The `out/` folder must exist before you use it with `-o`.

## Running the public tests

From this folder:

```
python tests/run_public_tests.py
```

The output lists every test and ends with a summary per grading group (A–H).
A failing test prints what it expected and what it got. Passing every public
test does not guarantee full points: the final grade uses a larger hidden test
set with different data.

## Submitting

Commit and push your work, then submit your repository URL on Moodle before
the deadline. The four `.py` files and `AI_USAGE.md` must be in the
repository root. Late submissions are not accepted, and only commits pushed
before the deadline are graded.

The `.gitignore` keeps generated files (`db.json`, `errors.log`,
`response_*.json`) out of your repository.

def load_module(name):

def run_cli(args, cwd):

def read_json(path):
    with open(path, encoding="utf-8") as f:
        return json.load(f)


def expected_lines(name):
    return (EXPECTED / name).read_text(encoding="utf-8").splitlines()


def graded_log(path):
    """The graded part of every WARNING line in a log file."""
    out = []
    for line in Path(path).read_text(encoding="utf-8").splitlines():
        m = WARNING_RE.match(line)
        if m:
            out.append(m.group(1))
    return out


def graded_message(message):
    """The graded part of a message returned by parse_files()."""
    return message.split(" | ", 1)[0].rstrip(" |")


def ids(entries):
    return [r["flight_id"] for r in entries]


class TempDirTest(unittest.TestCase):

    def assertCleanExit(self, proc, code):
        self.assertNotIn("Traceback", proc.stderr, "the program crashed with a traceback:\n" + proc.stderr)
        self.assertEqual(proc.returncode, code,
                         f"expected exit code {code}, got {proc.returncode}\nstderr:\n{proc.stderr}")


# ---------------------------------------------------------------- A. Structure

class A_Structure(unittest.TestCase):
    FILES = ["flight_parser.py", "validator.py", "csv_loader.py", "query.py", "AI_USAGE.md"]

    def test_required_files_exist(self):
        for name in self.FILES:
            with self.subTest(file=name):
                self.assertTrue((ROOT / name).is_file(), f"{name} is missing from the project root")

    def test_ai_usage_not_empty(self):
        path = ROOT / "AI_USAGE.md"
        self.assertTrue(path.is_file() and path.read_text(encoding="utf-8").strip(),
                        "AI_USAGE.md is missing or empty")

    def test_imports_are_silent(self):
        proc = subprocess.run(
            [sys.executable, "-c", "import flight_parser, validator, csv_loader, query"],
            cwd=str(ROOT), capture_output=True, text=True, timeout=TIMEOUT)
        self.assertEqual(proc.returncode, 0, "importing your modules failed:\n" + proc.stderr)
        self.assertEqual(proc.stdout + proc.stderr, "", "importing your modules must not print anything")

    def test_required_functions_exist(self):
        required = {
            "flight_parser": ["main"],
            "validator": ["load_airports", "validate_row", "to_record"],
            "csv_loader": ["parse_files"],
            "query": ["run_query", "run_queries"],
        }
        for module, funcs in required.items():
            mod = load_module(module)
            for func in funcs:
                with self.subTest(function=f"{module}.{func}"):
                    self.assertTrue(callable(getattr(mod, func, None)), f"{module}.{func} is missing")

    def test_student_constants(self):
        fp = load_module("flight_parser")
        placeholders = {"STUDENT_ID": "000RDB000", "FIRST_NAME": "Firstname", "LAST_NAME": "Lastname"}
        for name, placeholder in placeholders.items():
            with self.subTest(constant=name):
                value = getattr(fp, name, None)
                self.assertIsInstance(value, str, f"{name} must be a string")
                self.assertTrue(value and value.isascii(), f"{name} must be non-empty plain ASCII")
                self.assertNotRegex(value, r"[\s_]", f"{name} must not contain spaces or underscores")
                self.assertNotEqual(value, placeholder, f"set {name} to your own details")

    def test_standard_library_only(self):
        local = {p.stem for p in ROOT.glob("*.py")} | {p.name for p in ROOT.iterdir() if (p / "__init__.py").is_file()}
        allowed = set(sys.stdlib_module_names) | local
        for path in ROOT.glob("*.py"):
            tree = ast.parse(path.read_text(encoding="utf-8"))
            for node in ast.walk(tree):
                names = []
                if isinstance(node, ast.Import):
                    names = [a.name for a in node.names]
                elif isinstance(node, ast.ImportFrom) and node.level == 0 and node.module:
                    names = [node.module]
                for n in names:
                    with self.subTest(file=path.name, module=n):
                        self.assertIn(n.split(".")[0], allowed, f"{path.name} imports non-standard module {n}")

GOOD = ["BA2490", "LHR", "JFK", "2025-11-14 10:30", "2025-11-14 13:05", "489.99"]


def row(**changes):
    keys = ["flight_id", "origin", "destination", "dep", "arr", "price"]
    fields = list(GOOD)
    for k, v in changes.items():

    return fields

# ---------------------------------------------------------------- runner

class TallyResult(unittest.TextTestResult):
    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        self.passed = []

    def addSuccess(self, test):
        super().addSuccess(test)
        self.passed.append(test)


def main():
    os.chdir(ROOT)
    suite = unittest.defaultTestLoader.loadTestsFromModule(sys.modules[__name__])
    totals, passed = {}, {}
    for test in suite_tests(suite):
        group = type(test).__name__
        totals[group] = totals.get(group, 0) + 1

    runner = unittest.TextTestRunner(verbosity=2, resultclass=TallyResult)
    result = runner.run(suite)

    for test in result.passed:
        group = type(test).__name__
        passed[group] = passed.get(group, 0) + 1
    labels = {
        "A_Structure": "A. Structure", "B_RowValidation": "B. Row validation",
        "C_ParsingAFile": "C. Parsing a file", "D_CliAndFolders": "D. CLI and folders",
        "E_Duplicates": "E. Duplicates", "F_BasicQueries": "F. Basic queries",
        "G_ExtendedQueries": "G. Extended queries", "H_ResponseFile": "H. Response file",
    }
    print("\nSummary by grading group")
    for group in sorted(totals):
        print(f"  {labels.get(group, group):<24} {passed.get(group, 0)}/{totals[group]} tests passed")
    print("\nReminder: the final grade uses a larger hidden test set with different data.")
    return 0 if result.wasSuccessful() else 1


def suite_tests(suite):
    for item in suite:
        if isinstance(item, unittest.TestSuite):
            yield from suite_tests(item)
        else:
            yield item


if __name__ == "__main__":
    sys.exit(main())
